rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isSignedIn() { return request.auth != null; }
    function isOwner(uid) { return isSignedIn() && request.auth.uid == uid; }
    // GANTI 'GANTI_EMAIL_MASTER' di bawah supaya SAMA PERSIS dengan MASTER_EMAIL di js/config.js
    function isMaster() { return isSignedIn() && request.auth.token.email == 'GANTI_EMAIL_MASTER'; }

    // Profil tiap akun (nama usaha, dsb) — cuma pemiliknya sendiri yang boleh baca/tulis,
    // MASTER juga boleh BACA & HAPUS punya akun lain (dipakai fitur Backup & Hapus Akun di Admin)
    match /users/{uid} {
      allow read, write: if isOwner(uid);
      allow read, delete: if isMaster();
    }

    // Permintaan akses (register tahap 1) — siapa aja (belum login) boleh BIKIN,
    // tapi cuma master yang boleh BACA daftarnya & approve/tolak
    match /access_requests/{id} {
      allow create: if true;
      allow read, update, delete: if isMaster();
    }

    // Daftar email yang sudah di-approve — perlu dibaca saat register tahap 2 (belum login).
    // Master boleh create/delete bebas. User yang BARU register boleh update SEKALI buat
    // "klaim" invite jadi miliknya (usedBy = uid sendiri), tapi cuma kalau emailnya memang
    // udah di-approve master duluan (approved==true) — biar gak bisa dipakai sembarangan.
    match /allowed_emails/{email} {
      allow read: if true;
      allow create, delete: if isMaster();
      allow update: if isMaster()
        || (isSignedIn() && resource.data.approved == true && request.resource.data.usedBy == request.auth.uid);
    }

    // Semua data keuangan (Current Asset, Income, Expenses, Debts, Investment, Inventory/PPE/
    // Intangible + riwayat penjualannya, dst) — wajib login DAN dokumennya harus punya field
    // uid == uid yang lagi login, KECUALI master yang juga boleh baca/hapus data akun lain
    // (buat fitur Hapus Akun & Backup di Admin)
    match /{collection}/{docId} {
      allow read, update, delete: if collection in [
        'current_assets','accounts_receivable','inventory',
        'property_plant_equipment','intangible_assets','investments','investment_sales',
        'asset_sales',
        'income','expenses','debts','debt_payments',
        'payment_history','laporan_snapshots'
      ] && isSignedIn() && (resource.data.uid == request.auth.uid || isMaster());

      allow create: if collection in [
        'current_assets','accounts_receivable','inventory',
        'property_plant_equipment','intangible_assets','investments','investment_sales',
        'asset_sales',
        'income','expenses','debts','debt_payments',
        'payment_history','laporan_snapshots'
      ] && isSignedIn() && request.resource.data.uid == request.auth.uid;
    }
  }
}
